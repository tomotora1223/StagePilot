cmake_minimum_required(VERSION 3.22)
project(StagePilot VERSION 0.22.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(DEFINED JUCE_SOURCE_DIR)
  add_subdirectory("${JUCE_SOURCE_DIR}" JUCE)
else()
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 91ad83ae34a81e0833b1a2b0866f54846370ae53)
  FetchContent_MakeAvailable(JUCE)
endif()
juce_set_vst2_sdk_path("${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/FST")
juce_add_gui_app(StagePilot PRODUCT_NAME "StagePilot" COMPANY_NAME "StagePilot" VERSION 0.22.0)
juce_add_binary_data(StagePilotFonts SOURCES Assets/Fonts/ZenKakuGothicNew-Regular.ttf Assets/Fonts/ZenKakuGothicNew-Bold.ttf)
juce_generate_juce_header(StagePilot)
target_sources(StagePilot PRIVATE Source/Main.cpp Source/Engine.cpp Source/Tests.cpp Source/Sequence.cpp Source/Remote.cpp Source/RemoteTests.cpp Source/Recorder.cpp Source/RecordingTests.cpp Source/VideoRecorder.cpp Source/VideoTests.cpp Source/FxChain.cpp Source/FxTests.cpp Source/ReuseTests.cpp ThirdParty/QRCode/qrcodegen.cpp)
target_compile_definitions(StagePilot PRIVATE JUCE_ASIO=1 JUCE_PLUGINHOST_VST=1 JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1)
target_link_libraries(StagePilot PRIVATE StagePilotFonts juce::juce_audio_utils juce::juce_dsp juce::juce_recommended_config_flags)
if(WIN32)
  target_link_libraries(StagePilot PRIVATE mfplat mfreadwrite mfuuid mf ole32 oleaut32)
endif()
if(MSVC)
  target_link_options(StagePilot PRIVATE /DELAYLOAD:mfplat.dll /DELAYLOAD:mfreadwrite.dll /DELAYLOAD:mf.dll)
  target_link_libraries(StagePilot PRIVATE delayimp)
  target_compile_options(StagePilot PRIVATE /utf-8 /W4)
endif()
